/* 2. Web Audio Klangschale */
let audioCtx = null;
let masterGain = null;
let droneOscs = [];
let audioActive = false;

function initAudio() {
  if (audioCtx) return;
  const AudioContext = window.AudioContext || window.webkitAudioContext;
  audioCtx = new AudioContext();
  masterGain = audioCtx.createGain();
  masterGain.gain.setValueAtTime(0, audioCtx.currentTime);
  masterGain.connect(audioCtx.destination);

  const baseFreqs = [108, 216, 324, 432];
  baseFreqs.forEach((freq, idx) => {
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'sine';
    osc.frequency.setValueAtTime(freq + (Math.random() * 0.8 - 0.4), audioCtx.currentTime);
    gain.gain.setValueAtTime(0.08 / (idx + 1), audioCtx.currentTime);
    osc.connect(gain);
    gain.connect(masterGain);
    osc.start();
    droneOscs.push(osc);
  });
}

function toggleAudio() {
  initAudio();
  if (audioCtx.state === 'suspended') audioCtx.resume();

  const btnText = document.getElementById('audioText');
  const t = I18N[currentLang];
  if (!audioActive) {
    masterGain.gain.setTargetAtTime(0.35, audioCtx.currentTime, 2.5);
    audioActive = true;
    btnText.textContent = t.audioActive;
    playSingingBowl(216);
  } else {
    masterGain.gain.setTargetAtTime(0, audioCtx.currentTime, 1.5);
    audioActive = false;
    btnText.textContent = t.audioIdle;
  }
}

function playSingingBowl(freq = 216) {
  if (!audioCtx || !audioActive) return;
  const now = audioCtx.currentTime;
  const harmonics = [1, 2.01, 3.02, 4.18];
  harmonics.forEach((h, i) => {
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'sine';
    osc.frequency.setValueAtTime(freq * h, now);
    const attack = 0.02;
    const decay = 4.5 / (i + 1);
    gain.gain.setValueAtTime(0, now);
    gain.gain.linearRampToValueAtTime(0.25 / (i + 1), now + attack);
    gain.gain.exponentialRampToValueAtTime(0.0001, now + decay);
    osc.connect(gain);
    gain.connect(masterGain);
    osc.start(now);
    osc.stop(now + decay);
  });
}

document.getElementById('audioBtn').onclick = toggleAudio;

const cv = document.getElementById('sky'), cx = cv.getContext('2d');
const still = matchMedia('(prefers-reduced-motion:reduce)').matches;
let W, H, stars = [], shoots = [], nextShoot = 2000;
const tints = ['255,255,255', '255,255,255', '255,240,200', '200,225,255'];

function initSky() {
  W = cv.width = innerWidth; H = cv.height = innerHeight;
  stars = Array.from({length: Math.round(W * H / 5500)}, () => ({
    x: Math.random() * W, y: Math.random() * H, r: Math.random() * 1.3 + .2,
    p: Math.random() * 6.28, s: Math.random() * 1.6 + .4,
    c: tints[Math.floor(Math.random() * tints.length)]
  }));
}

function spawnShoot() {
  const a = (Math.random() * .5 + .3), dir = Math.random() < .5 ? 1 : -1;
  shoots.push({
    x: Math.random() * W * (dir > 0 ? .7 : 1), y: Math.random() * H * .4,
    vx: Math.cos(a) * dir * (10 + Math.random() * 6), vy: Math.sin(a) * (10 + Math.random() * 6),
    life: 0, max: 40 + Math.random() * 25
  });
}

function frameSky(t) {
  cx.clearRect(0, 0, W, H);
  for (const s of stars) {
    const k = still ? .7 : .5 + .5 * Math.sin(t / 1000 * s.s + s.p);
    const al = .2 + .8 * k;
    cx.fillStyle = `rgba(${s.c},${al})`;
    cx.beginPath(); cx.arc(s.x, s.y, s.r * (.8 + .3 * k), 0, 6.283); cx.fill();
  }
  if (!still) {
    if (t > nextShoot) { spawnShoot(); nextShoot = t + 3000 + Math.random() * 5000; }
    shoots = shoots.filter(s => s.life < s.max);
    for (const s of shoots) {
      s.x += s.vx; s.y += s.vy; s.life++;
      const f = 1 - s.life / s.max, tx = s.x - s.vx * 6, ty = s.y - s.vy * 6;
      const g = cx.createLinearGradient(s.x, s.y, tx, ty);
      g.addColorStop(0, `rgba(255,255,255,${f})`); g.addColorStop(1, 'rgba(180,200,255,0)');
      cx.strokeStyle = g; cx.lineWidth = 1.6; cx.lineCap = 'round';
      cx.beginPath(); cx.moveTo(s.x, s.y); cx.lineTo(tx, ty); cx.stroke();
    }
    requestAnimationFrame(frameSky);
  }
}
